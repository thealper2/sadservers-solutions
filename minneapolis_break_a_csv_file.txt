#!/bin/bash

for i in {1..10}
do
    filename=$(printf "data-%02d.csv" $((i-1)))
    
    echo -n "$filename: "
    
    {
        head -1 data.csv
        tail -n +2 data.csv | cut -d "," -f $i
    } > "$filename"
    
    size=$(stat -c%s "$filename")
    if [ $size -gt 32768 ]; then
        truncate --size=32768 "$filename"
        echo "$size bytes -> TRUNCATED"
    else
        echo "$size bytes ✓"
    fi
done

echo ""
echo "=== FILE INFORMATION ==="
ls -lh data-*.csv

echo ""
echo "=== CHECK SCRIPT ==="
/home/admin/agent/check.sh
