#!/bin/bash

# Configuration
LOG_DIR="/var/log/app"
DAYS=7

echo "Starting Cleanup..."

find "$LOG_DIR" -maxdepth 1 -name "*.log" -type f -mtime +$DAYS -print -delete

echo "Cleanup finished."
